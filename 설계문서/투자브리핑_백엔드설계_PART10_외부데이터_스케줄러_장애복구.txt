문서 목적
------------------------------------------------------------
이 문서는 투자 브리핑 시스템의 외부 시장데이터 수집과 스케줄러 운영구조를
실제 Java 클래스 단위로 상세 설계한다.

주요 대상
1. KRX 데이터 수집
2. 해외 시세 수집
3. 환율 수집
4. 채권금리 수집
5. 거래일·휴일·주말 처리
6. 재시도·타임아웃·Rate Limit
7. Circuit Breaker
8. API 로그와 데이터 검증
9. PostgreSQL UPSERT
10. 스케줄러 실행순서와 ShedLock
11. 장애복구와 수동 재실행 API
12. 통합 테스트

기술 기준
- Java 21
- Spring Boot 3.5.7
- Spring WebClient
- Resilience4j
- Spring Scheduling
- ShedLock
- PostgreSQL
- Spring Data JPA / QueryDSL
- Flyway
- Testcontainers / WireMock

핵심 원칙
- 외부 API 응답을 즉시 계산에 사용하지 않는다.
- 수집 → 검증 → 정규화 → 저장 → 최신성 확인 후 계산한다.
- 외부 응답 DTO와 내부 표준 DTO를 분리한다.
- 실패 데이터로 마지막 정상 데이터를 덮어쓰지 않는다.
- 외부 API 호출 중 DB 트랜잭션을 오래 유지하지 않는다.

@Configuration
public class WebClientConfig {

    @Bean
    public WebClient.Builder webClientBuilder() {
        HttpClient httpClient = HttpClient.create()
            .option(ChannelOption.CONNECT_TIMEOUT_MILLIS, 5_000)
            .responseTimeout(Duration.ofSeconds(15))
            .doOnConnected(connection -> connection
                .addHandlerLast(new ReadTimeoutHandler(15))
                .addHandlerLast(new WriteTimeoutHandler(15))
            );

        return WebClient.builder()
            .clientConnector(new ReactorClientHttpConnector(httpClient))
            .defaultHeader(HttpHeaders.ACCEPT, MediaType.APPLICATION_JSON_VALUE);
    }
}

11. PostgreSQL UPSERT
============================================================

업무키
- TB_PRC_DAY: BASE_DT + STK_ID
- TB_IDX_DAY: BASE_DT + IDX_ID
- TB_EXCH_DAY: BASE_DT + CURR_PAIR_CD
- TB_BOND_DAY: BASE_DT + BOND_CD
- TB_MKT_SNAP: BASE_DT + MKT_SNAP_CD

예시

INSERT INTO TB_PRC_DAY (
    BASE_DT, STK_ID, OPEN_PRC, HIGH_PRC, LOW_PRC,
    CLOSE_PRC, VOL, TRD_AMT, DATA_SRC_CD,
    DATA_STS, COLLECT_DTTM
)
VALUES (
    :baseDate, :stockId, :openPrice, :highPrice,
    :lowPrice, :closePrice, :volume, :tradingAmount,
    :dataSourceCode, :dataStatus, :collectedDateTime
)
ON CONFLICT (BASE_DT, STK_ID)
DO UPDATE SET
    OPEN_PRC = EXCLUDED.OPEN_PRC,
    HIGH_PRC = EXCLUDED.HIGH_PRC,
    LOW_PRC = EXCLUDED.LOW_PRC,
    CLOSE_PRC = EXCLUDED.CLOSE_PRC,
    VOL = EXCLUDED.VOL,
    TRD_AMT = EXCLUDED.TRD_AMT,
    DATA_SRC_CD = EXCLUDED.DATA_SRC_CD,
    DATA_STS = EXCLUDED.DATA_STS,
    COLLECT_DTTM = EXCLUDED.COLLECT_DTTM;

대량 저장
- NamedParameterJdbcTemplate.batchUpdate 권장
- 종목별 JpaRepository.save 반복 비권장

        max-attempts: 3
        wait-duration: 2s

  circuitbreaker:
    instances:
      krxApi:
        sliding-window-type: COUNT_BASED
        sliding-window-size: 20
        minimum-number-of-calls: 10
        failure-rate-threshold: 50
        wait-duration-in-open-state: 60s
        permitted-number-of-calls-in-half-open-state: 3

  ratelimiter:
    instances:
      krxApi:
        limit-for-period: 5
        limit-refresh-period: 1s
        timeout-duration: 2s

@Component
@RequiredArgsConstructor
public class MarketDataCollectionJob {

    private final MarketDataCollectionService collectionService;
    private final SchedulerExecutionService schedulerExecutionService;

    @Scheduled(
        cron = "${investment.scheduler.market-data.cron}",
        zone = "Asia/Seoul"
    )
    @SchedulerLock(
        name = "marketDataCollectionJob",
        lockAtMostFor = "PT30M",
        lockAtLeastFor = "PT1M"
    )
    public void execute() {
        schedulerExecutionService.execute(
            "MARKET_DATA_COLLECT",
            LocalDate.now(ZoneId.of("Asia/Seoul")),
            collectionService::collectAll
        );
    }
}

17. ShedLock
============================================================

의존성
implementation 'net.javacrumbs.shedlock:shedlock-spring:6.3.0'
implementation 'net.javacrumbs.shedlock:shedlock-provider-jdbc-template:6.3.0'

설정

@Configuration
@EnableScheduling
@EnableSchedulerLock(defaultLockAtMostFor = "PT30M")
public class SchedulingConfig {

    @Bean
    public LockProvider lockProvider(DataSource dataSource) {
        return new JdbcTemplateLockProvider(
            JdbcTemplateLockProvider.Configuration.builder()
                .withJdbcTemplate(new JdbcTemplate(dataSource))
                .usingDbTime()
                .build()
        );
    }
}

DDL

CREATE TABLE SHEDLOCK (
    NAME       VARCHAR(64)  NOT NULL,
    LOCK_UNTIL TIMESTAMP(3) NOT NULL,
    LOCKED_AT  TIMESTAMP(3) NOT NULL,
    LOCKED_BY  VARCHAR(255) NOT NULL,
    CONSTRAINT PK_SHEDLOCK PRIMARY KEY (NAME)
);

COMMENT ON TABLE SHEDLOCK IS '분산 스케줄러 중복실행 방지 잠금';

18. SchedulerExecutionService
============================================================

@Service
@RequiredArgsConstructor
public class SchedulerExecutionService {

    private final SchedulerLogService schedulerLogService;

    public void execute(
        String jobCode,
        LocalDate baseDate,
        Runnable job
    ) {
        String traceId = UUID.randomUUID().toString();
        Long schedulerLogId = schedulerLogService.start(
            traceId,
            jobCode,
            baseDate
        );

        long startedAt = System.nanoTime();

        try {
            job.run();
            schedulerLogService.success(
                schedulerLogId,
                elapsedMillis(startedAt)
            );
        } catch (PartialCollectionException exception) {
            schedulerLogService.partialSuccess(
                schedulerLogId,
                exception.getSuccessCount(),
                exception.getFailedCount(),
                elapsedMillis(startedAt),
                exception.getMessage()
            );
        } catch (Exception exception) {
            schedulerLogService.failure(
                schedulerLogId,
                elapsedMillis(startedAt),
                exception.getMessage()
            );
            throw exception;
        }
    }
}

24. MarketDataCollectionService 예시
============================================================

@Service
@RequiredArgsConstructor
public class MarketDataCollectionService {

    private final TradingCalendarService tradingCalendarService;
    private final KrxMarketCollector krxMarketCollector;
    private final KrxStockPriceCollector krxStockPriceCollector;
    private final OverseasPriceCollector overseasPriceCollector;
    private final ExchangeRateCollector exchangeRateCollector;
    private final BondYieldCollector bondYieldCollector;

    public MarketDataCollectionSummary collectAll() {
        LocalDate today = LocalDate.now(ZoneId.of("Asia/Seoul"));

        LocalDate koreaTradeDate =
            tradingCalendarService.resolveLatestTradingDay(
                today.minusDays(1),
                MarketCalendarType.KOREA
            );

        LocalDate usTradeDate =
            tradingCalendarService.resolveLatestTradingDay(
                today.minusDays(1),
                MarketCalendarType.UNITED_STATES
            );

        CollectionResult<MarketSnapshotData> krMarket =
            krxMarketCollector.collect(koreaTradeDate);

        CollectionResult<StockDailyPriceData> krPrices =
            krxStockPriceCollector.collect(
                koreaTradeDate,
                loadDomesticStockCodes()
            );

        CollectionResult<StockDailyPriceData> usPrices =
            overseasPriceCollector.collect(
                usTradeDate,
                loadOverseasTickers()
            );

        CollectionResult<ExchangeRateData> exchangeRates =
            exchangeRateCollector.collect(koreaTradeDate);

        CollectionResult<BondYieldData> bondYields =
            bondYieldCollector.collect(usTradeDate);

        return MarketDataCollectionSummary.of(
            krMarket,
            krPrices,
            usPrices,
            exchangeRates,
            bondYields
        );
    }
}
